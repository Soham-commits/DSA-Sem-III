Program 1: Matrix Addition
Source File: Add.c

Description:
Performs element-wise addition of two matrices entered by
the user. Both matrices must have the same dimensions.
------------------------------------------------------------

1.  SOURCE CODE
------------------------------------------------------------
#include<stdio.h>

void AcceptMatrix(int m, int n, int MatA[m][n])
{
    for(int i=0; i<n; i++)
    {
        for(int j=0; j<m; j++)
        {
            scanf("%d", &MatA[i][j]);
        }
    }
}

void PrintMatrix(int m, int n, int MatA[m][n])
{
    for(int i = 0; i < m; i++)
    {
        for(int j = 0; j < n; j++)
        {
            printf("%d\t", MatA[i][j]);
        }
        printf("\n");
    }
}


void AdditionMatrix(int m, int n, int MatA[m][n], int MatB[m][n], int MatC[m][n])
{
    printf("The addition of Matrix Elements are:  ");

    for(int i=0; i<n; i++)
    {
        for(int j=0; j<m; j++)
        {
            MatC[i][j] = MatA[i][j] + MatB[i][j];
        }
    }

}

int main()
{
    int m =0, n=0;
    printf("Enter the Matrix row count: ");
    scanf("%d", &m);

    printf("Enter the Matrix column count: ");
    scanf("%d", &n);

    int MatA[m][n];
    int MatB[m][n];
    int MatC[m][n];


    printf("Enter first matrix elements:\n");
    AcceptMatrix(m, n, MatA);

    printf("Enter second matrix elements:\n");
    AcceptMatrix(m, n, MatB);

    AdditionMatrix(m, n, MatA, MatB, MatC);

    printf("The addition of Matrix Elements are:\n");
    PrintMatrix(m, n, MatC);
    return 0;
}

#include<stdio.h>

{
    for(int i=0; i<n; i++)

void SubstractionMatrix(int m, int n, int MatA[m][n], int MatB[m][n], int MatC[m][n])
{

    AcceptMatrix(m, n, MatB);

    SubstractionMatrix(n,m,MatA,MatB,MatC);

    PrintMatrix(m, n, MatC);

Program 3: Matrix Multiplication
Source File: Multiplcation.c

Description:
Performs multiplication of two matrices. The number of columns
in the first matrix must equal the number of rows in the second
matrix for multiplication to be possible.
------------------------------------------------------------

1.  SOURCE CODE
------------------------------------------------------------
#include <stdio.h>

void AcceptMatrix(int m, int n, int MatA[m][n])
{
    for(int i = 0; i < m; i++)
    {
        for(int j = 0; j < n; j++)
        {
            scanf("%d", &MatA[i][j]);
        }
    }
}

void PrintMatrix(int m, int n, int MatA[m][n])
{
    for(int i = 0; i < m; i++)
    {
        for(int j = 0; j < n; j++)
        {
            printf("%d\t", MatA[i][j]);
        }
        printf("\n");
    }
}

void MultiplicationMatrix(int r1, int c1, int c2,
                          int A[r1][c1],
                          int B[c1][c2],
                          int C[r1][c2])
{
    for(int i = 0; i < r1; i++)
    {
        for(int j = 0; j < c2; j++)
        {
            C[i][j] = 0;
            for(int k = 0; k < c1; k++)
            {
                C[i][j] += A[i][k] * B[k][j];
            }
        }
    }
}

int main()
{
    int r1, c1, r2, c2;

    printf("Enter rows and columns of first matrix: ");
    scanf("%d %d", &r1, &c1);

    printf("Enter rows and columns of second matrix: ");
    scanf("%d %d", &r2, &c2);

    if(c1 != r2)
    {
        printf("Multiplication not possible\n");
        return 0;
    }

    int MatA[r1][c1];
    int MatB[r2][c2];
    int MatC[r1][c2];

    printf("Enter first matrix elements:\n");
    AcceptMatrix(r1, c1, MatA);

    printf("Enter second matrix elements:\n");
    AcceptMatrix(r2, c2, MatB);

    MultiplicationMatrix(r1, c1, c2, MatA, MatB, MatC);

    printf("The Multiplication of Matrix Elements are:\n");
    PrintMatrix(r1, c2, MatC);

    return 0;
}

Program 4: Matrix Transpose
Source File: Transpose.c

Description:
Computes the transpose of a matrix (rows become columns and
columns become rows).
------------------------------------------------------------

1.  SOURCE CODE
------------------------------------------------------------
#include <stdio.h>

void AcceptMatrix(int m, int n, int MatA[m][n])
{
    for(int i = 0; i < m; i++)
    {
        for(int j = 0; j < n; j++)
        {
            scanf("%d", &MatA[i][j]);
        }
    }
}

void PrintMatrix(int m, int n, int MatA[m][n])
{
    for(int i = 0; i < m; i++)
    {
        for(int j = 0; j < n; j++)
        {
            printf("%d\t", MatA[i][j]);
        }
        printf("\n");
    }
}

void Transpose(int m, int n, int MatA[m][n], int MatB[n][m])
{
    for(int i = 0; i < m; i++)
    {
        for(int j = 0; j < n; j++)
        {
            MatB[j][i] = MatA[i][j];
        }
    }
}

int main()
{
    int m, n;

    printf("Enter the Matrix row count: ");
    scanf("%d", &m);

    printf("Enter the Matrix column count: ");
    scanf("%d", &n);

    int MatA[m][n];
    int MatB[n][m];

    printf("Enter The matrix elements:\n");
    AcceptMatrix(m, n, MatA);

    Transpose(m, n, MatA, MatB);

    printf("The Transpose Matrix Elements are:\n");
    PrintMatrix(n, m, MatB);

    return 0;
}
